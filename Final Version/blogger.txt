        const CONFIG = {
            apiUrl: "https://smart-novel-api.vercel.app/api/search",
            googleScript: "https://script.google.com/macros/s/AKfycbyzThZAU6jwIl7PeYlJL3Swu61b8TCS9F7gHwNGoVh7hEXPd9zBRGhb7tUwmWDwC-Wj/exec",
            perPage: 21
        };

        const RESTRICTED_KEYWORDS = [
            "bold", "sexy", "sex", "18+", "adult", "dirty",
            "kiss", "ganda", "gandi", "galat","suhagrat","tharki","porn","nude","husna kanwal","husnay kanwal"
        ];

        let state = { 
            searchResults: [], 
            totalResults: 0,
            currentPage: 1, 
            favorites: JSON.parse(localStorage.getItem("favorites") || "[]"), 
            showFavorites: false,
            lastLog: "",
            randomOffset: Math.floor(Math.random() * 75000) 
        };

        // --- TYPEWRITER PLACEHOLDER LOGIC ---
        const typingTexts = [
            "Konsa novel parhna hai aaj?...",
            "Boring din? Ek acha novel search karein!...",
            "Apna pasandeeda novel search karein..."
        ];
        let typeTextIdx = 0;
        let charIdx = 0;
        let isDeleting = false;
        let typewriterTimeout;
        let isFocused = false; 
        let isListening = false; 

        function typeWriterPlaceholder() {
            const searchInput = document.getElementById("searchBox");
            
            if (isFocused || searchInput.value.trim() !== "" || isListening) {
                clearTimeout(typewriterTimeout);
                return;
            }

            const currentText = typingTexts[typeTextIdx];
            
            if (isDeleting) {
                searchInput.setAttribute("placeholder", currentText.substring(0, charIdx - 1));
                charIdx--;
            } else {
                searchInput.setAttribute("placeholder", currentText.substring(0, charIdx + 1));
                charIdx++;
            }

            let typingSpeed = isDeleting ? 40 : 80;

            if (!isDeleting && charIdx === currentText.length) {
                typingSpeed = 2000; 
                isDeleting = true;
            } else if (isDeleting && charIdx === 0) {
                isDeleting = false;
                typeTextIdx = (typeTextIdx + 1) % typingTexts.length;
                typingSpeed = 500; 
            }

            typewriterTimeout = setTimeout(typeWriterPlaceholder, typingSpeed);
        }

        // --- SMART CLEAR BUTTON LOGIC ---
        function toggleClearBtn() {
            const btn = document.getElementById("clearBtn");
            const box = document.getElementById("searchBox");
            if(box.value.trim() !== "") {
                btn.classList.add("show-btn");
            } else {
                btn.classList.remove("show-btn");
            }
        }

        // --- KEYBOARD NAVIGATION STATE ---
        let currentFocus = -1;

        (function init() { showSkeletonLoading(); fetchData(); setupEventListeners(); typeWriterPlaceholder(); toggleClearBtn(); })();

        // --- SECURITY FUNCTION ---
        function escapeHTML(str) {
            if (!str) return "";
            return str.replace(/[&<>"']/g, function(m) {
                switch (m) {
                    case '&': return '&amp;';
                    case '<': return '&lt;';
                    case '>': return '&gt;';
                    case '"': return '&quot;';
                    case "'": return '&#039;';
                    default: return m;
                }
            });
        }

        function setupEventListeners() {
            const searchBox = document.getElementById("searchBox");
            
            searchBox.addEventListener("input", (e) => {
                toggleClearBtn(); 
                debounce(() => showSuggestions(e.target.value), 300)();
            });
            
            // KEYBOARD NAVIGATION LOGIC
            searchBox.addEventListener("keydown", (e) => { 
                const box = document.getElementById("suggestions");
                let items = box.getElementsByClassName("suggestion-item");
                
                if (e.key === "ArrowDown") {
                    currentFocus++;
                    addActive(items);
                    e.preventDefault(); 
                } else if (e.key === "ArrowUp") {
                    currentFocus--;
                    addActive(items);
                    e.preventDefault();
                } else if (e.key === "Enter") {
                    e.preventDefault();
                    if (currentFocus > -1 && box.classList.contains("active")) {
                        if (items[currentFocus]) {
                            items[currentFocus].click();
                        }
                    } else {
                        box.classList.remove("active"); 
                        searchNovel(); 
                    }
                }
            });

            document.addEventListener("click", (e) => { 
                if (!document.querySelector(".search-container").contains(e.target)) { 
                    document.getElementById("suggestions").classList.remove("active"); 
                } 
            });
            
            searchBox.addEventListener("focus", () => {
                isFocused = true;
                clearTimeout(typewriterTimeout);
                searchBox.setAttribute("placeholder", "Search novel title..."); 
            });
            
            searchBox.addEventListener("blur", () => {
                isFocused = false;
                if (searchBox.value.trim() === "" && !isListening) {
                    charIdx = 0;
                    isDeleting = false;
                    typeWriterPlaceholder(); 
                }
            });
        }

        function addActive(items) {
            if (!items) return false;
            removeActive(items);
            if (currentFocus >= items.length) currentFocus = 0;
            if (currentFocus < 0) currentFocus = (items.length - 1);
            
            if(items[currentFocus].classList.contains('no-hover')) {
                currentFocus = -1;
                return;
            }
            
            items[currentFocus].classList.add("selected");
            items[currentFocus].scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        }

        function removeActive(items) {
            for (let i = 0; i < items.length; i++) {
                items[i].classList.remove("selected");
            }
        }

        // --- FETCH DATA FROM VERCEL API ---
        async function fetchData(query = "") {
            if (!navigator.onLine) { document.getElementById("results").innerHTML = "<p style='text-align:center'>Please check your internet.</p>"; return; }
            showSkeletonLoading();

            let start = (state.currentPage - 1) * CONFIG.perPage;
            
            if (!query && !state.showFavorites) {
                start += state.randomOffset;
            }

            let url = `${CONFIG.apiUrl}?offset=${start}`;
            
            if (query) {
                url = `${CONFIG.apiUrl}?query=${encodeURIComponent(query)}`;
            }

            try {
                const response = await fetch(url);
                const result = await response.json();
                
                state.totalResults = result.total || 0;
                state.searchResults = result.data || [];
                
                renderResults();
            } catch (err) {
                console.error("API Fetch Error:", err);
                document.getElementById("results").innerHTML = "<p style='text-align:center'>Server Connection Error. Please try again later.</p>";
            }
        }

        function searchNovel() {
            const queryRaw = document.getElementById("searchBox").value.trim();
            const query = queryRaw.toLowerCase();
            
            document.getElementById("searchBox").blur();

                } else {
                    resultCountSpan.style.visibility = "visible";
                    resultCountSpan.textContent = "Found Novels: 0";
                    grid.innerHTML = `
                    <div style="grid-column: 1/-1; text-align: center; padding: 40px; display: flex; flex-direction: column; align-items: center; justify-content: center;">
                        <h3 style="font-size: 1.5rem; margin-bottom: 10px; font-style: italic; color: var(--ink);">No novels found 😕</h3>

                dataToRender = state.favorites.slice((state.currentPage - 1) * CONFIG.perPage, state.currentPage * CONFIG.perPage);

                dataToRender = state.searchResults.slice((state.currentPage - 1) * CONFIG.perPage, state.currentPage * CONFIG.perPage);

                let safeTitle = escapeHTML(novel.Titles);

                        const regex = new RegExp(`(${safeQuery})`, 'gi');
                        safeTitle = safeTitle.replace(regex, '<mark class="highlight">$1</mark>');
                    } catch(e) { }
                }

        async function showSuggestions(query) {
            const box = document.getElementById("suggestions");
            currentFocus = -1; 

            if (!query) { box.classList.remove("active"); return; }
            const isRestricted = RESTRICTED_KEYWORDS.some(word => query.toLowerCase().includes(word));
            if (isRestricted) { box.classList.remove("active"); return; }
            
            try {
                const res = await fetch(`${CONFIG.apiUrl}?query=${encodeURIComponent(query)}`);
                const result = await res.json();
                
                const suggestionsList = (result.data || []).slice(0, 5);
                
                if (suggestionsList.length === 0) { 
                    box.classList.remove("active");
                    return; 
                }
                
                box.innerHTML = suggestionsList.map(r => {
                    const safeTitle = escapeHTML(r.Titles);
                    const jsSafeTitle = r.Titles.replace(/'/g, "\\'"); 
                    
                    let highlightedTitle = safeTitle;
                    try {
                        const safeQuery = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
                        const regex = new RegExp(`(${safeQuery})`, 'gi');
                        highlightedTitle = safeTitle.replace(regex, '<span style="color: var(--ink); font-weight: 800;">$1</span>');
                    } catch(e) {}

                    return `<div class="suggestion-item" onclick="selectSuggestion('${jsSafeTitle}')">
                                <span class="suggestion-icon">
                                    <svg focusable="false" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" width="18" height="18" fill="currentColor"><path d="M15.5 14h-.79l-.28-.27A6.471 6.471 0 0 0 16 9.5 6.5 6.5 0 1 0 9.5 16c1.61 0 3.09-.59 4.23-1.57l.27.28v.79l5 4.99L20.49 19l-4.99-5zm-6 0C7.01 14 5 11.99 5 9.5S7.01 5 9.5 5 14 7.01 14 9.5 11.99 14 9.5 14z"></path></svg>
                                </span>
                                <span>${highlightedTitle}</span>
                            </div>`;
                }).join("");
                
                box.classList.add("active");
            } catch(e) {
                box.classList.remove("active");
            }
        }

        function toggleFav(title, link, btn) {
            const idx = state.favorites.findIndex(f => f.Titles === title);
            if (idx === -1) { state.favorites.push({ Titles: title, Links: link }); btn.classList.add("active"); btn.textContent = "\u2764\ufe0f"; showToast("Saved to favorites"); } 
            else { state.favorites.splice(idx, 1); btn.classList.remove("active"); btn.textContent = "\uD83E\uDD0D"; showToast("Removed from favorites"); if(state.showFavorites) { renderResults(); } }
            localStorage.setItem("favorites", JSON.stringify(state.favorites));
        }

        function showToast(msg) {
            const container = document.getElementById("toastContainer");

        function shareNovel(title, link) {
            if (navigator.share) navigator.share({ title: title, text: `Read ${title}`, url: link });
            else { navigator.clipboard.writeText(link); showToast("Link copied!"); }
        }

        function renderPagination(totalPages) {
            const div = document.getElementById("pagination");
            div.innerHTML = "";
            if (totalPages <= 1) return;
            const { currentPage } = state;
            const createBtn = (page, text, isNav = false, isDisabled = false) => {
                const btn = document.createElement("button");
                btn.className = `page-btn ${isNav ? 'nav-btn' : ''} ${page === currentPage && !isNav ? 'active' : ''}`;
                btn.textContent = text || page;
                btn.disabled = isDisabled;
                if (!isDisabled) { 
                    btn.onclick = () => { 
                        state.currentPage = page; 
                        if (state.showFavorites) { renderResults(); } 
                        else { fetchData(document.getElementById("searchBox").value.trim()); }
                        document.getElementById("novel-app-wrapper").scrollIntoView({ behavior: 'smooth' }); 
                    }; 
                }
                return btn;
            };
            const createEllipsis = () => { const span = document.createElement("span"); span.className = "pagination-ellipsis"; span.innerHTML = "&hellip;"; return span; };

            div.appendChild(createBtn(currentPage - 1, "Prev", true, currentPage === 1));
            const pages = []; pages.push(1);
            let start = Math.max(2, currentPage - 1); let end = Math.min(totalPages - 1, currentPage + 1);
            if (start > 2) pages.push('...');
            for (let i = start; i <= end; i++) { if (i > 1 && i < totalPages) pages.push(i); }
            if (end < totalPages - 1) pages.push('...');
            if (totalPages > 1) pages.push(totalPages);
            pages.forEach(page => { if (page === '...') div.appendChild(createEllipsis()); else div.appendChild(createBtn(page)); });
            div.appendChild(createBtn(currentPage + 1, "Next", true, currentPage === totalPages));
        }

        function debounce(func, wait) { let timeout; return function(...args) { clearTimeout(timeout); timeout = setTimeout(() => func.apply(this, args), wait); }; }
        
        function startVoiceSearch() {
            const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!SR) { showToast("Voice search not supported on this browser."); return; }
            const r = new SR();
            
            r.lang = 'en-US'; 
            
            r.onstart = () => { 
                isListening = true; 
                clearTimeout(typewriterTimeout); 
                
                document.getElementById("searchBox").value = ""; 
                toggleClearBtn(); 
                document.getElementById("suggestions").classList.remove("active");

                document.getElementById("searchBox").placeholder = "Listening... Speak now"; 
                document.querySelector(".mic").classList.add("listening"); 
            };
            
            r.onend = () => { 
                isListening = false; 
                document.getElementById("searchBox").placeholder = "Search novel title..."; 
                document.querySelector(".mic").classList.remove("listening"); 
                
                if (document.getElementById("searchBox").value.trim() === "" && !isFocused) {
                    charIdx = 0;
                    isDeleting = false;
                    typeWriterPlaceholder();
                }
            };

            r.onresult = (e) => { 
                const transcript = e.results[0][0].transcript; 
                document.getElementById("searchBox").value = transcript; 
                toggleClearBtn(); 
                searchNovel(); 
            };

            r.onerror = (event) => {
                if (event.error === 'no-speech' || event.error === 'audio-capture') {
                    showToast("Didn't catch that. Please try again.");
                } else if (event.error === 'not-allowed') {
                    showToast("Microphone access denied.");
                }
            };

            r.start();
        }

        function getUserId() {
            let id = localStorage.getItem("nb_user_id");
            if (!id) { id = 'user_' + Date.now().toString(36) + Math.random().toString(36).substr(2, 5); localStorage.setItem("nb_user_id", id); }
            return id;
        }

  function logSearch(q, n) { 
    const userId = getUserId();
    const deviceType = /Mobi|Android|iPhone/i.test(navigator.userAgent) ? "Mobile" : "Desktop";
    
    const resultStatus = n ? "Yes" : "No";

    const params = new URLSearchParams({ 
        search: q, 
        timestamp: new Date().toISOString(), 
        noResults: resultStatus,
        device: deviceType, 
        userId: userId 
    });
    
    fetch(CONFIG.googleScript, { method: 'POST', body: params }).catch(e => {}); 
}
