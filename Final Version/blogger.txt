    <script>
        const CONFIG = {
            supabaseUrl: "https://qcpjvngjwbulkkpdxouq.supabase.co",
            supabaseKey: "sb_publishable_dCy-rNGFvPGhzbfhLmhvyg_wKzxmeFF",
            perPage: 21,
            googleScript: "https://script.google.com/macros/s/AKfycbzuJc2bicMhDKbrBhaSQj3tKakoihMosV9F2j-HIelvyOp4YIra0fr7V7glc7tLF-N1/exec"
        };

        const RESTRICTED_KEYWORDS = ["bold", "sexy", "sex", "18+", "adult", "dirty", "kiss", "ganda", "gandi", "galat", "suhagrat", "tharki", "porn", "nude", "husna kanwal", "husnay kanwal"];

        let state = { 
            searchResults: [], 
            currentPage: 1, 
            totalFound: 0,
            favorites: JSON.parse(localStorage.getItem("favorites") || "[]"), 
            showFavorites: false 
        };

        (function init() { 
            fetchData(); 
            setupEventListeners(); 
        })();

        function setupEventListeners() {
            const searchBox = document.getElementById("searchBox");
            searchBox.addEventListener("keydown", (e) => { if (e.key === "Enter") { searchNovel(); } });
        }

        // --- FETCH DATA FROM SUPABASE ---
        async function fetchData(query = "") {
            showSkeletonLoading();
            const start = (state.currentPage - 1) * CONFIG.perPage;
            const end = start + CONFIG.perPage - 1;

            let url = `${CONFIG.supabaseUrl}/rest/v1/novels?select=*`;
            
            if (query) {
                // Roman Urdu/English ke liye 'ilike' (Case-insensitive) search
                url += `&Titles=ilike.*${encodeURIComponent(query)}*`;
            }

            try {
                const response = await fetch(url, {
                    headers: {
                        "apikey": CONFIG.supabaseKey,
                        "Authorization": `Bearer ${CONFIG.supabaseKey}`,
                        "Range-Unit": "items",
                        "Range": `${start}-${end}`,
                        "Prefer": "count=exact"
                    }
                });

                const data = await response.json();
                
                // Content-Range header se total records nikalna
                const contentRange = response.headers.get("content-range");
                state.totalFound = contentRange ? parseInt(contentRange.split('/')[1]) : data.length;
                state.searchResults = data;

                renderResults();
            } catch (err) {
                console.error("Supabase Error:", err);
                document.getElementById("results").innerHTML = "<p style='grid-column:1/-1; text-align:center;'>Unable to connect to the novel bank. Please refresh.</p>";
            }
        }

        function searchNovel() {
            const queryRaw = document.getElementById("searchBox").value.trim();
            const query = queryRaw.toLowerCase();
            
            // Safety Checks
            if (RESTRICTED_KEYWORDS.some(word => query.includes(word))) {
                showRestrictionMessage();
                return;
            }

            const urduPattern = /[\u0600-\u06FF]/;
            if (urduPattern.test(queryRaw)) {
                showUrduWarning();
                return;
            }

            state.showFavorites = false;
            state.currentPage = 1;
            fetchData(queryRaw);
            
            // Log search
            logSearch(queryRaw, false);
        }

        function renderResults() {
            const grid = document.getElementById("results");
            grid.innerHTML = "";
            
            const total = state.showFavorites ? state.favorites.length : state.totalFound;
            document.getElementById("resultCount").textContent = `Library: ${total} Novels Available`;

            const dataToRender = state.showFavorites ? state.favorites : state.searchResults;

            if (dataToRender.length === 0) {
                grid.innerHTML = `<div style="grid-column:1/-1; text-align:center; padding:40px;"><h3>No novels found 😕</h3><button class="btn-go-back" onclick="clearSearch()">⬅ Return to Library</button></div>`;
                document.getElementById("pagination").innerHTML = "";
                return;
            }

            dataToRender.forEach(novel => {
                const isFav = state.favorites.some(f => f.Titles === novel.Titles);
                const jsSafeTitle = novel.Titles.replace(/'/g, "\\'");
                const safeLink = (novel.Links && novel.Links.startsWith('http')) ? novel.Links : '#';

                grid.innerHTML += `
                    <article class="novel-card">
                        <h2 class="novel-title">${novel.Titles}</h2>
                        <div class="card-actions">
                            <button class="btn btn-secondary btn-fav ${isFav ? 'active' : ''}" onclick="toggleFav('${jsSafeTitle}', '${safeLink}', this)">${isFav ? '❤️' : '🤍'}</button>
                            <button class="btn btn-secondary" onclick="shareNovel('${jsSafeTitle}', '${safeLink}')">🔗</button>
                            <a href="${safeLink}" target="_blank" class="btn btn-download"><span>📥</span> Download PDF</a>
                        </div>
                    </article>`;
            });

            if (!state.showFavorites) {
                renderPagination(Math.ceil(state.totalFound / CONFIG.perPage));
            } else {
                document.getElementById("pagination").innerHTML = "";
            }
        }

        function renderPagination(totalPages) {
            const div = document.getElementById("pagination");
            div.innerHTML = "";
            if (totalPages <= 1) return;

            const createBtn = (text, targetPage, isDisabled) => {
                const btn = document.createElement("button");
                btn.className = "page-btn";
                btn.textContent = text;
                btn.disabled = isDisabled;
                btn.onclick = () => { 
                    state.currentPage = targetPage; 
                    fetchData(document.getElementById("searchBox").value.trim()); 
                    window.scrollTo({top: 0, behavior: 'smooth'});
                };
                return btn;
            };

            div.appendChild(createBtn("Previous", state.currentPage - 1, state.currentPage === 1));
            
            const info = document.createElement("span");
            info.className = "page-info";
            info.textContent = `Page ${state.currentPage} of ${totalPages}`;
            div.appendChild(info);

            div.appendChild(createBtn("Next", state.currentPage + 1, state.currentPage === totalPages));
        }

        function toggleFav(title, link, btn) {
            const idx = state.favorites.findIndex(f => f.Titles === title);
            if (idx === -1) {
                state.favorites.push({ Titles: title, Links: link });
                btn.classList.add("active"); btn.textContent = "❤️";
                showToast("Saved to favorites");
            } else {
                state.favorites.splice(idx, 1);
                btn.classList.remove("active"); btn.textContent = "🤍";
                showToast("Removed from favorites");
            }
            localStorage.setItem("favorites", JSON.stringify(state.favorites));
            if (state.showFavorites) renderResults();
        }

        function toggleFavorites() {
            state.showFavorites = !state.showFavorites;
            const btn = document.getElementById("favToggle");
            if (state.showFavorites) {
                btn.textContent = "Show All Library";
                btn.style.background = "var(--secondary)";
                btn.style.color = "white";
            } else {
                btn.textContent = "Show Favorites ❤️";
                btn.style.background = "white";
                btn.style.color = "var(--secondary)";
            }
            renderResults();
        }

        function clearSearch() {
            document.getElementById("searchBox").value = "";
            state.currentPage = 1;
            fetchData();
        }

        function showSkeletonLoading() {
            const grid = document.getElementById("results");
            grid.innerHTML = "";
            for(let i=0; i<6; i++) {
                grid.innerHTML += `<div class="novel-card"><div class="skeleton skeleton-text"></div><div class="skeleton skeleton-text" style="width:60%"></div><div class="skeleton skeleton-btn"></div></div>`;
            }
        }

        function showToast(msg) {
            const container = document.getElementById("toastContainer");
            const toast = document.createElement("div"); toast.className = "toast"; toast.textContent = msg; 
            container.appendChild(toast);
            setTimeout(() => toast.classList.add("show"), 10);
            setTimeout(() => { toast.classList.remove("show"); setTimeout(() => toast.remove(), 300); }, 2500);
        }

        function shareNovel(title, link) {
            if (navigator.share) navigator.share({ title: title, url: link });
            else { navigator.clipboard.writeText(link); showToast("Link copied to clipboard!"); }
        }

        function startVoiceSearch() {
            const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!SR) { showToast("Voice search not supported"); return; }
            const r = new SR();
            r.onstart = () => { document.getElementById("searchBox").placeholder = "Listening..."; document.querySelector(".mic").classList.add("listening"); };
            r.onend = () => { document.getElementById("searchBox").placeholder = "Search novel title..."; document.querySelector(".mic").classList.remove("listening"); };
            r.onresult = (e) => { document.getElementById("searchBox").value = e.results[0][0].transcript; searchNovel(); };
            r.start();
        }

        function showUrduWarning() {
            document.getElementById("results").innerHTML = `<div style="grid-column:1/-1; text-align:center; padding:40px; background:#eff6ff; border-radius:12px; border:1px solid #3b82f6;"><h3 style="color:#1d4ed8;">Please Use English Letters</h3><p>Example: Type <b>"Namal"</b> instead of <b>"نمل"</b></p><button class="btn-go-back" onclick="clearSearch()">Try Again</button></div>`;
        }

        function showRestrictionMessage() {
            document.getElementById("results").innerHTML = `<div style="grid-column:1/-1; text-align:center; padding:40px; background:#fee2e2; border-radius:12px; border:1px solid #ef4444;"><h3 style="color:#b91c1c;">Restricted Search</h3><p>We maintain a family-friendly novel bank.</p><button class="btn-go-back" onclick="clearSearch()" style="background:#dc2626;">Clear Search</button></div>`;
        }

        function logSearch(q, n) {
            const params = new URLSearchParams({ search: q, timestamp: new Date().toISOString(), noResults: n });
            fetch(CONFIG.googleScript, { method: 'POST', body: params }).catch(e => {}); 
        }
    </script>
