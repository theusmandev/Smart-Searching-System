    <script>
        const CONFIG = {
            supabaseUrl: "https://qcpjvngjwbulkkpdxouq.supabase.co",
            supabaseKey: "sb_publishable_dCy-rNGFvPGhzbfhLmhvyg_wKzxmeFF",
            googleScript: "https://script.google.com/macros/s/AKfycbzuJc2bicMhDKbrBhaSQj3tKakoihMosV9F2j-HIelvyOp4YIra0fr7V7glc7tLF-N1/exec",
            perPage: 21
        };

        const RESTRICTED_KEYWORDS = ["bold", "sexy", "sex", "18+", "adult", "dirty", "kiss", "ganda", "gandi", "galat","suhagrat","tharki","porn","nude","husna kanwal","husnay kanwal"];

        let state = { 
            searchResults: [], 
            totalResults: 0,
            currentPage: 1, 
            favorites: JSON.parse(localStorage.getItem("favorites") || "[]"), 
            showFavorites: false 
        };

        (function init() { fetchData(); setupEventListeners(); })();

        function escapeHTML(str) {
            if (!str) return "";
            return str.replace(/[&<>"']/g, m => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m]));
        }

        function setupEventListeners() {
            const searchBox = document.getElementById("searchBox");
            searchBox.addEventListener("input", debounce((e) => showSuggestions(e.target.value), 300));
            searchBox.addEventListener("keydown", (e) => { if (e.key === "Enter") { document.getElementById("suggestions").classList.remove("active"); searchNovel(); } });
            document.addEventListener("click", (e) => { if (!document.querySelector(".search-container").contains(e.target)) { document.getElementById("suggestions").classList.remove("active"); } });
        }

        async function fetchData(query = "") {
            showSkeletonLoading();
            const start = (state.currentPage - 1) * CONFIG.perPage;
            const end = start + CONFIG.perPage - 1;

            let url = `${CONFIG.supabaseUrl}/rest/v1/novels?select=*`;
            if (query) {
                url += `&Titles=ilike.*${encodeURIComponent(query)}*`;
            }

            try {
                const response = await fetch(url, {
                    headers: {
                        "apikey": CONFIG.supabaseKey,
                        "Authorization": `Bearer ${CONFIG.supabaseKey}`,
                        "Range-Unit": "items",
                        "Range": `${start}-${end}`,
                        "Prefer": "count=exact"
                    }
                });

                const data = await response.json();
                const contentRange = response.headers.get("content-range");
                state.totalResults = contentRange ? parseInt(contentRange.split('/')[1]) : data.length;
                state.searchResults = data;
                renderResults();
            } catch (err) {
                console.error(err);
                document.getElementById("results").innerHTML = "<p style='text-align:center'>Connection Error. Please check your internet.</p>";
            }
        }

        function searchNovel() {
            const queryRaw = document.getElementById("searchBox").value.trim();
            const query = queryRaw.toLowerCase();
            document.getElementById("searchBox").blur();
            document.getElementById("suggestions").classList.remove("active");

            if (RESTRICTED_KEYWORDS.find(word => query.includes(word))) {
                showSecurityUI("restricted"); return;
            }
            if (/[\u0600-\u06FF]/.test(queryRaw)) {
                showSecurityUI("urdu"); return;
            }
            if (query !== "" && query.replace(/\s/g, '').length < 3) {
                showSecurityUI("short"); return;
            }

            state.currentPage = 1;
            state.showFavorites = false;
            fetchData(queryRaw);
            logSearch(queryRaw, false);
        }

        function renderResults() {
            const grid = document.getElementById("results");
            document.getElementById("recommendedSection").style.display = "none";
            grid.innerHTML = "";
            
            const dataToRender = state.showFavorites ? state.favorites : state.searchResults;
            const total = state.showFavorites ? state.favorites.length : state.totalResults;
            
            document.getElementById("resultCount").textContent = `${state.showFavorites ? 'Favorite' : 'Found'} Novels: ${total}`;

            if (total === 0) {
                grid.innerHTML = `<div style="grid-column: 1/-1; text-align: center; padding: 40px;"><h3 style="font-size: 1.5rem; margin-bottom: 10px;">No novels found 😕</h3><button class="btn-go-back" onclick="clearSearch()">⬅ Go Back to Library</button></div>`;
                document.getElementById("pagination").innerHTML = ""; return;
            }

            const searchQuery = document.getElementById("searchBox").value.trim();

            dataToRender.forEach(novel => {
                const isFav = state.favorites.some(f => f.Titles === novel.Titles);
                let safeTitle = escapeHTML(novel.Titles);
                
                if (searchQuery && searchQuery.length >= 3 && !state.showFavorites) {
                    const regex = new RegExp(`(${searchQuery.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
                    safeTitle = safeTitle.replace(regex, '<mark class="highlight">$1</mark>');
                }

                const jsSafeTitle = novel.Titles.replace(/'/g, "\\'"); 
                const safeLink = novel.Links || '#';

                grid.innerHTML += `
                <article class="novel-card">
                    <h2 class="novel-title" title="${escapeHTML(novel.Titles)}">${safeTitle}</h2>
                    <div class="card-actions">
                        <button class="btn btn-secondary btn-fav ${isFav ? 'active' : ''}" onclick="toggleFav('${jsSafeTitle}', '${safeLink}', this)">${isFav ? '❤️' : '🤍'}</button>
                        <button class="btn btn-secondary" onclick="shareNovel('${jsSafeTitle}', '${safeLink}')">🔗</button>
                        <a href="${safeLink}" target="_blank" class="btn btn-download"><span>📥</span> Download PDF</a>
                    </div>
                </article>`;
            });

            if (!state.showFavorites) renderPagination(Math.ceil(total / CONFIG.perPage));
        }

        function renderPagination(totalPages) {
            const div = document.getElementById("pagination");
            div.innerHTML = "";
            if (totalPages <= 1) return;

            const createBtn = (page, text, isNav = false, isDisabled = false) => {
                const btn = document.createElement("button");
                btn.className = `page-btn ${isNav ? 'nav-btn' : ''} ${page === state.currentPage && !isNav ? 'active' : ''}`;
                btn.textContent = text || page;
                btn.disabled = isDisabled;
                btn.onclick = () => { state.currentPage = page; fetchData(document.getElementById("searchBox").value.trim()); window.scrollTo({top: 0, behavior: 'smooth'}); };
                return btn;
            };

            div.appendChild(createBtn(state.currentPage - 1, "Prev", true, state.currentPage === 1));
            
            // Simple logic for current page info
            const span = document.createElement("span");
            span.textContent = `Page ${state.currentPage} of ${totalPages}`;
            span.style.margin = "0 10px";
            div.appendChild(span);

            div.appendChild(createBtn(state.currentPage + 1, "Next", true, state.currentPage === totalPages));
        }

        function showSecurityUI(type) {
            const grid = document.getElementById("results");
            const configs = {
                restricted: { bg: '#fee2e2', border: '#ef4444', icon: '🛡️', title: 'Search Restricted', p: 'Search for decent Urdu novels only.' },
                urdu: { bg: '#eff6ff', border: '#3b82f6', icon: '⌨️', title: 'Use English / Roman Urdu', p: 'Example: Type "Jannat" instead of "جنت"' },
                short: { bg: '#fef3c7', border: '#d97706', icon: '🤔', title: 'Search Too Short', p: 'Type at least 3 characters.' }
            };
            const c = configs[type];
            grid.innerHTML = `<div style="grid-column: 1/-1; text-align: center; padding: 40px; background: ${c.bg}; border-radius: 12px; border: 1px solid ${c.border};">
                <div style="font-size: 3rem;">${c.icon}</div><h3>${c.title}</h3><p>${c.p}</p>
                <button class="btn-go-back" onclick="clearSearch()">Try Again</button>
            </div>`;
            document.getElementById("pagination").innerHTML = "";
        }

        async function showSuggestions(query) {
            const box = document.getElementById("suggestions");
            if (!query || query.length < 2) { box.classList.remove("active"); return; }
            
            const url = `${CONFIG.supabaseUrl}/rest/v1/novels?Titles=ilike.*${encodeURIComponent(query)}*&select=Titles&limit=5`;
            try {
                const res = await fetch(url, { headers: {"apikey": CONFIG.supabaseKey, "Authorization": `Bearer ${CONFIG.supabaseKey}`} });
                const data = await res.json();
                if (data.length === 0) { box.classList.remove("active"); return; }
                box.innerHTML = data.map(item => `<div class="suggestion-item" onclick="selectSuggestion('${item.Titles.replace(/'/g, "\\'")}')">${escapeHTML(item.Titles)}</div>`).join("");
                box.classList.add("active");
            } catch(e) {}
        }

        function selectSuggestion(title) { document.getElementById("searchBox").value = title; searchNovel(); }

        function toggleFav(title, link, btn) {
            const idx = state.favorites.findIndex(f => f.Titles === title);
            if (idx === -1) { state.favorites.push({ Titles: title, Links: link }); btn.classList.add("active"); btn.textContent = "❤️"; showToast("Saved"); } 
            else { state.favorites.splice(idx, 1); btn.classList.remove("active"); btn.textContent = "🤍"; showToast("Removed"); }
            localStorage.setItem("favorites", JSON.stringify(state.favorites));
            if(state.showFavorites) renderResults();
        }

        function toggleFavorites() {
            state.showFavorites = !state.showFavorites;
            const btn = document.getElementById("favToggle");
            btn.textContent = state.showFavorites ? "Show All Library" : "Show Favorites";
            state.currentPage = 1;
            state.showFavorites ? renderResults() : fetchData();
        }

        function showSkeletonLoading() {
            const grid = document.getElementById("results");
            grid.innerHTML = Array(6).fill(`<div class="novel-card"><div class="skeleton skeleton-text"></div><div class="skeleton skeleton-text" style="width: 60%"></div><div class="skeleton skeleton-btn"></div></div>`).join("");
        }

        function showToast(msg) {
            const container = document.getElementById("toastContainer");
            const toast = document.createElement("div"); toast.className = "toast"; toast.textContent = msg; container.appendChild(toast);
            setTimeout(() => toast.classList.add("show"), 10);
            setTimeout(() => { toast.classList.remove("show"); setTimeout(() => toast.remove(), 300); }, 2000);
        }

        function clearSearch() { document.getElementById("searchBox").value = ""; state.currentPage = 1; fetchData(); }
        function debounce(f, w) { let t; return (...a) => { clearTimeout(t); t = setTimeout(() => f(...a), w); }; }
        function shareNovel(t, l) { if (navigator.share) navigator.share({ title: t, url: l }); else { navigator.clipboard.writeText(l); showToast("Link copied!"); } }
        function startVoiceSearch() {
            const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!SR) { showToast("Not supported"); return; }
            const r = new SR();
            r.onstart = () => { document.getElementById("searchBox").placeholder = "Listening..."; document.querySelector(".mic").classList.add("listening"); };
            r.onend = () => { document.getElementById("searchBox").placeholder = "Search..."; document.querySelector(".mic").classList.remove("listening"); };
            r.onresult = (e) => { document.getElementById("searchBox").value = e.results[0][0].transcript; searchNovel(); };
            r.start();
        }
        function logSearch(q, n) { fetch(CONFIG.googleScript, { method: 'POST', body: new URLSearchParams({ search: q, noResults: n }) }).catch(e => {}); }
    </script>
